A simple example for plotting a graph

```python
def plot_figure(values, figure_name, label="f(g)", show = False):
    fig, ax = plt.subplots(figsize=(8, 5))        # size in inches
    ax.plot(np.arange(len(values)), values, label=label)
    ax.set_title(figure_name)
    ax.set_xlabel("Iteration")
    ax.set_xticks(np.arange(len(values))) # optional if you want the x interval to be integers
    ax.set_ylabel("Objective value")
    ax.legend()                                   # uses the label= from plot()
    ax.grid(True)
    fig.tight_layout()
    fig.savefig(f"{figure_name}.png", dpi=150)    # optional
    if show:
        plt.show()
    plt.close(fig) # optional if you want to close the fig
```
